📊 API de Datos — SafePark

Responsable: Edgar  |  Puerto: 3002  |  Tecnología: Node.js + Express + MySQL

1. Descripción

Esta API expone endpoints REST para consultar áreas verdes, reportes, eventos y estadísticas generales del sistema SafePark. Se conecta a la base de datos safepark_db en MySQL.

2. Requisitos previos

  • Node.js instalado (nodejs.org)
  • XAMPP corriendo con MySQL activo
  • Base de datos safepark_db ya creada

3. Instalación

Abrir terminal en la raíz del proyecto y ejecutar:

mkdir API_Datos
cd API_Datos
npm init -y
npm install express mysql2 cors dotenv

4. Archivo de configuración (.env)

Crear el archivo API_Datos/.env con:

DB_HOST=localhost
DB_USER=root
DB_PASSWORD=
DB_NAME=safepark_db
PORT=3002
⚠️ No subir el archivo .env a GitHub. Agregar .env al .gitignore.

5. Código principal (index.js)

Crear el archivo API_Datos/index.js:

const express = require('express');
const mysql = require('mysql2/promise');
const cors = require('cors');
require('dotenv').config();

const app = express();
app.use(cors());
app.use(express.json());

const pool = mysql.createPool({
    host: process.env.DB_HOST,
    user: process.env.DB_USER,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_NAME
});

// GET /areas — todas las áreas con coordenadas
app.get('/areas', async (req, res) => {
    const [rows] = await pool.query(
        'SELECT * FROM AREA WHERE lat IS NOT NULL AND lng IS NOT NULL ORDER BY id_area DESC'
    );
    res.json(rows);
});

// GET /areas/:id — detalle de un área
app.get('/areas/:id', async (req, res) => {
    const [rows] = await pool.query(
        'SELECT * FROM AREA WHERE id_area = ?', [req.params.id]
    );
    if (rows.length === 0) return res.status(404).json({ error: 'No encontrada' });
    res.json(rows[0]);
});

// GET /areas/:id/reportes — reportes de un área específica
app.get('/areas/:id/reportes', async (req, res) => {
    const [rows] = await pool.query(
        'SELECT * FROM REPORTE WHERE id_area = ? ORDER BY fecha DESC',
        [req.params.id]
    );
    res.json(rows);
});

// GET /estadisticas — resumen general del sistema
app.get('/estadisticas', async (req, res) => {
    const [[{ total_areas }]]    = await pool.query('SELECT COUNT(*) as total_areas FROM AREA');
    const [[{ total_usuarios }]] = await pool.query('SELECT COUNT(*) as total_usuarios FROM USUARIO');
    const [[{ total_reportes }]] = await pool.query('SELECT COUNT(*) as total_reportes FROM REPORTE');
    const [[{ total_eventos }]]  = await pool.query('SELECT COUNT(*) as total_eventos FROM EVENTO');
    res.json({ total_areas, total_usuarios, total_reportes, total_eventos });
});

// GET /reportes — reportes recientes con nombre de usuario y área
app.get('/reportes', async (req, res) => {
    const limite = parseInt(req.query.limite) || 20;
    const [rows] = await pool.query(
        `SELECT R.*, U.nombre as usuario, A.nombre as area
         FROM REPORTE R
         JOIN USUARIO U ON R.id_usuario = U.id_usuario
         JOIN AREA A ON R.id_area = A.id_area
         ORDER BY R.fecha DESC LIMIT ?`,
        [limite]
    );
    res.json(rows);
});

// GET /eventos — todos los eventos con creador y área
app.get('/eventos', async (req, res) => {
    const [rows] = await pool.query(
        `SELECT E.*, U.nombre as creador, A.nombre as area
         FROM EVENTO E
         JOIN USUARIO U ON E.id_usuario = U.id_usuario
         LEFT JOIN AREA A ON E.id_area = A.id_area
         ORDER BY E.fecha ASC`
    );
    res.json(rows);
});

app.listen(process.env.PORT, () =>
    console.log('API Datos corriendo en puerto ' + process.env.PORT)
);

6. Cómo correr la API

cd API_Datos
node index.js

Deberías ver: API Datos corriendo en puerto 3002

7. Endpoints disponibles

MétodoRutaDescripción
GET/areasLista todas las áreas con coordenadas
GET/areas/:idDetalle de un área por ID
GET/areas/:id/reportesReportes de un área específica
GET/estadisticasTotales generales del sistema
GET/reportesReportes recientes (con ?limite=N)
GET/eventosTodos los eventos próximos

8. Probar la API

Abrir el navegador o usar Postman:

http://localhost:3002/estadisticas
http://localhost:3002/areas
http://localhost:3002/reportes?limite=5
💡 El parámetro ?limite=N en /reportes controla cuántos resultados devuelve. Por defecto son 20.
