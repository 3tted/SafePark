🧑‍💻 API de Usuarios — SafePark

Responsable: Arleth  |  Puerto: 3001  |  Tecnología: Node.js + Express + MySQL

1. Descripción

Esta API expone endpoints REST para gestionar los usuarios de SafePark. Se conecta directamente a la base de datos safepark_db en MySQL y sirve datos al frontend PHP o a otras APIs del proyecto.

2. Requisitos previos

  • Node.js instalado (nodejs.org)
  • XAMPP corriendo con MySQL activo
  • Base de datos safepark_db ya creada (pedir a Edgar o al equipo)

3. Instalación

Abrir terminal en la raíz del proyecto y ejecutar:

mkdir API_Usuarios
cd API_Usuarios
npm init -y
npm install express mysql2 cors dotenv bcrypt

4. Archivo de configuración (.env)

Crear el archivo API_Usuarios/.env con:

DB_HOST=localhost
DB_USER=root
DB_PASSWORD=
DB_NAME=safepark_db
PORT=3001
⚠️ No subir el archivo .env a GitHub. Agregar .env al .gitignore.

5. Código principal (index.js)

Crear el archivo API_Usuarios/index.js:

const express = require('express');
const mysql = require('mysql2/promise');
const cors = require('cors');
require('dotenv').config();

const app = express();
app.use(cors());
app.use(express.json());

const pool = mysql.createPool({
    host: process.env.DB_HOST,
    user: process.env.DB_USER,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_NAME
});

// GET /usuarios — lista todos los usuarios
app.get('/usuarios', async (req, res) => {
    const [rows] = await pool.query(
        'SELECT id_usuario, nombre, correo, rol, fecha_registro FROM USUARIO'
    );
    res.json(rows);
});

// GET /usuarios/:id — perfil de un usuario
app.get('/usuarios/:id', async (req, res) => {
    const [rows] = await pool.query(
        'SELECT id_usuario, nombre, correo, rol, foto_perfil, fecha_registro FROM USUARIO WHERE id_usuario = ?',
        [req.params.id]
    );
    if (rows.length === 0) return res.status(404).json({ error: 'No encontrado' });
    res.json(rows[0]);
});

// PUT /usuarios/:id — actualizar nombre o rol
app.put('/usuarios/:id', async (req, res) => {
    const { nombre, rol } = req.body;
    await pool.query(
        'UPDATE USUARIO SET nombre = ?, rol = ? WHERE id_usuario = ?',
        [nombre, rol, req.params.id]
    );
    res.json({ ok: true });
});

// DELETE /usuarios/:id — eliminar usuario
app.delete('/usuarios/:id', async (req, res) => {
    await pool.query('DELETE FROM USUARIO WHERE id_usuario = ?', [req.params.id]);
    res.json({ ok: true });
});

// GET /usuarios/:id/reportes — reportes del usuario
app.get('/usuarios/:id/reportes', async (req, res) => {
    const [rows] = await pool.query(
        'SELECT * FROM REPORTE WHERE id_usuario = ? ORDER BY fecha DESC',
        [req.params.id]
    );
    res.json(rows);
});

// GET /usuarios/:id/favoritos — áreas favoritas del usuario
app.get('/usuarios/:id/favoritos', async (req, res) => {
    const [rows] = await pool.query(
        `SELECT A.* FROM FAVORITO F
         JOIN AREA A ON F.id_area = A.id_area
         WHERE F.id_usuario = ?`,
        [req.params.id]
    );
    res.json(rows);
});

app.listen(process.env.PORT, () =>
    console.log('API Usuarios corriendo en puerto ' + process.env.PORT)
);

6. Cómo correr la API

cd API_Usuarios
node index.js

Deberías ver: API Usuarios corriendo en puerto 3001
